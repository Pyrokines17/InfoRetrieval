#16
select ord.onum, sal.sname, cust.cname
from ord
         join sal on ord.snum = sal.snum
         join cust on ord.cnum = cust.cnum
where sal.city in ('London', 'Rome')
order by ord.onum;

#17
select sal.sname, sum(ord.amt), sum(sal.comm*ord.amt)
from ord
         join sal on ord.snum = sal.snum
where substring(ord.odate, 1, 2) <= '05'
group by sal.sname
order by sal.sname;

#18
select ord.onum, ord.amt, sal.sname, cust.cname
from ord
         join sal on ord.snum = sal.snum
         join cust on ord.cnum = cust.cnum
where (sal.city like 'L%' or sal.city like 'M%' or sal.city like 'N%' or sal.city like 'O%' or sal.city like 'P%' or sal.city like 'Q%' or sal.city like 'R%')
  and (cust.city like 'L%' or cust.city like 'M%' or cust.city like 'N%' or cust.city like 'O%' or cust.city like 'P%' or cust.city like 'Q%' or cust.city like 'R%');
